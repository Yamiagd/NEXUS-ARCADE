        try { return JSON.parse(localStorage.getItem('playhub_users')) || {}; } catch { return {}; }
    }
    function saveUsers(users) { localStorage.setItem('playhub_users', JSON.stringify(users)); }
    function getCurrentUser() { return localStorage.getItem('playhub_current_user') || null; }
    function setCurrentUser(email) { if (email) localStorage.setItem('playhub_current_user', email); else localStorage.removeItem('playhub_current_user'); }

    function updateBadge() {
        const user = getCurrentUser();
        userBadge.textContent = user ? '👤 ' + (user.length > 20 ? user.slice(0,18)+'…' : user) : '👤 guest';

    function handleAuth() {
        const email = authEmail.value.trim().toLowerCase();
        const provider = authProvider.value;
        const password = authPassword.value.trim();
        const confirm = authConfirm.value.trim();
        if (!email || !password) { showError('Please fill in all fields'); return; }
        if (!email.includes('@') || !email.includes('.')) { showError('Enter a valid email'); return; }
        if (password.length < 4) { showError('Password min 4 chars'); return; }
        const users = getUsers();
        if (isLoginMode) {
            if (!users[email]) { showError('No account found'); return; }
            if (users[email].password !== password) { showError('Incorrect password'); return; }
            setCurrentUser(email);
            enterApp(email);
        } else {
            if (users[email]) { showError('Account exists. Login instead.'); return; }
            if (password !== confirm) { showError('Passwords do not match'); return; }
            users[email] = { password, provider, created: new Date().toISOString() };
            saveUsers(users);
            setCurrentUser(email);
            enterApp(email);
        }
    }
    authActionBtn.addEventListener('click', handleAuth);
    authEmail.addEventListener('keydown', e => { if (e.key === 'Enter') handleAuth(); });
    authPassword.addEventListener('keydown', e => { if (e.key === 'Enter') handleAuth(); });
    authConfirm.addEventListener('keydown', e => { if (e.key === 'Enter') handleAuth(); });

    }

    });

        chatInitialized = true;
        window.chatInitialized = true;
        const messagesDiv = document.getElementById('chatMessages');
        const input = document.getElementById('chatInput');
        const sendBtn = document.getElementById('chatSend');
        const userCountSpan = document.getElementById('userCount');
        const base = email.split('@')[0];
        chatUsername = base + Math.floor(Math.random() * 1000);
        const channel = new BroadcastChannel('playhub_chat');
        chatChannel = channel;

        function loadHistory() {
            const stored = localStorage.getItem('chatHistory');
            if (stored) { try { JSON.parse(stored).forEach(m => appendMessage(m.user, m.text, m.time, false)); } catch(e) {} }
        }
        function appendMessage(user, text, time, save = true) {
            const div = document.createElement('div');
            div.className = 'msg';
            const timeStr = time || new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
            div.innerHTML = `<span class="user">${user}</span><span class="time">${timeStr}</span><br>${text}`;
            messagesDiv.appendChild(div);
            messagesDiv.scrollTop = messagesDiv.scrollHeight;
            if (save) {
                const stored = localStorage.getItem('chatHistory');
                let history = stored ? JSON.parse(stored) : [];
                history.push({ user, text, time: timeStr });
                if (history.length > 200) history = history.slice(-200);
                localStorage.setItem('chatHistory', JSON.stringify(history));
            }
        }
        function sendMessage(text) {
            if (!text.trim()) return;
            const msg = { type: 'message', user: chatUsername, text: text.trim(), time: new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'}) };
            channel.postMessage(msg);
            appendMessage(msg.user, msg.text, msg.time);
        }
        channel.onmessage = function(e) {
            const data = e.data;
            if (data.type === 'message') appendMessage(data.user, data.text, data.time);
            else if (data.type === 'ping') channel.postMessage({ type: 'pong', from: chatUsername });
            else if (data.type === 'pong') { activeUsers.add(data.from); updateUserCount(); }
        };
        sendBtn.addEventListener('click', () => { sendMessage(input.value); input.value = ''; });
        input.addEventListener('keydown', (e) => { if (e.key === 'Enter') { sendMessage(input.value); input.value = ''; } });
        loadHistory();
        let activeUsers = new Set([chatUsername]);
        function updateUserCount() { userCountSpan.textContent = `👤 ${activeUsers.size} online`; }
        setInterval(() => { channel.postMessage({ type: 'ping', from: chatUsername }); }, 10000);
        setInterval(() => { if (activeUsers.size > 50) { activeUsers = new Set([chatUsername]); updateUserCount(); } }, 30000);
        setTimeout(() => { channel.postMessage({ type: 'message', user: '🟢 System', text: `${chatUsername} joined the chat`, time: new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'}) }); }, 500);
        updateUserCount();
    }

    // Auto-login check after loading
    window.addEventListener('load', function() {
        // If loading already finished and we have a user, init
        if (!loadingScreen.classList.contains('hidden')) return;
        const user = getCurrentUser();
        if (user) {
            const users = getUsers();
            if (users[user]) {
                authScreen.style.display = 'none';
                mainContent.style.display = 'flex';
                updateBadge();
                setTimeout(() => {
                    initChatWithUser(user);
                    if (!window.gamesInitialized) initGamesAndChat();
                    updateBadge();
                }, 200);
            }
        }
    });
